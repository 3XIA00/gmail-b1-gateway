# gmail-gateway — external, locked Python dependencies + runtime (R5).
#
# The package is SOURCE self-contained (no imports outside this tree + the
# stdlib + the pins below) but its third-party dependencies are EXTERNAL and must
# be provided by the verifier's environment. This file is that declaration so a
# third party can reproduce the results offline without guessing.
#
# requires-python: >=3.10,<3.15   (explicitly INCLUDING 3.14 — see P1 below)
#
# --require-hashes: EVERY requirement below is pinned (exact ==) and carries the
# PyPI-published sha256 of every wheel/sdist pip may select on EITHER declared
# runtime, so `pip install --require-hashes -r requirements.txt` installs a
# byte-identical dependency set or fails closed. The hashes are PyPI's own
# digests (not recomputed here); anyone can re-derive them from the JSON API URL
# printed above each package. Pure packages ship one wheel for all platforms;
# cryptography/cffi/hypothesis ship per-(python,OS,arch) wheels — the set below
# covers both runtime rows and the sdist fallback.
#
# Reproduce (clean env, from the payload root == this dir):
#   python -m venv .venv && . .venv/bin/activate      # or Scripts\activate on Windows
#   pip install --require-hashes -r requirements.txt
#   python -m pytest -q                                # -> 507 passed
#   python -m pytest authz/ -q                         # authz review surface only
#       -> 165 passed                                  # needs only cryptography + pytest
